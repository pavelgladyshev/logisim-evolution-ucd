# The protocol without gdb: PacketSize is the input buffer's size in hex (0x10000), vCont is not offered (gdb uses
# c and s), malformed packets get an error and unknown ones the empty reply, and the session goes on; Ctrl-C while
# the CPU runs gets the stop reply T02 (SIGINT)
^----- session raw$
+ ^\? -> S05$
+ PacketSize=10000;qXfer:features:read\+$
+ ^vCont\? -> \(empty\)$
+ ^Zx -> E01$
+ ^mzz,4 -> E00$
+ ^Z0,zz,4 -> E01$
+ ^qNoSuchQuery -> \(empty\)$
+ ^xNoSuchPacket -> \(empty\)$
+ ^p1041 -> 03000000$
+ ^c, Ctrl-C -> T02$
+ ^p1041 -> 03000000$

# a new gdb after the raw session closed without D; it dies at a breakpoint
^----- session gdb_crash.gdb$
^Breakpoint 1, square
^\$1 = 0$

# after the crash the CPU ran on, and its breakpoint is gone; kill ends the session
^----- session gdb_after_crash.gdb$
^\$1 = 1$
^Breakpoint 1, do_ecall \(\)
^\[Inferior 1 \(Remote target\) killed\]$

^\$2 = 1$
